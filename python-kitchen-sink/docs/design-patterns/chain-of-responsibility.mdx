Passes a request along a chain of handlers; each one either handles it, partly handles it, or forwards it to the next. Shows up in LLD rounds for ATM cash dispensers, logging levels, request validation/middleware pipelines, approval workflows and support ticket escalation.

## Syntax

```python
from abc import ABC, abstractmethod

class Handler(ABC):
    def __init__(self) -> None:
        self._next: Handler | None = None

    def set_next(self, handler: "Handler") -> "Handler":
        self._next = handler
        return handler                          # allows a.set_next(b).set_next(c)

    def handle(self, request: dict) -> str | None:
        if self._next:
            return self._next.handle(request)
        return None                             # end of chain

class AuthHandler(Handler):
    def handle(self, request: dict) -> str | None:
        if not request.get("token"):
            return "401 unauthorized"           # stop the chain
        return super().handle(request)          # pass along

class RateLimitHandler(Handler):
    def handle(self, request: dict) -> str | None:
        if request.get("calls", 0) > 100:
            return "429 too many requests"
        return super().handle(request)

class ValidationHandler(Handler):
    def handle(self, request: dict) -> str | None:
        if "body" not in request:
            return "400 bad request"
        return super().handle(request)

chain = AuthHandler()
chain.set_next(RateLimitHandler()).set_next(ValidationHandler())

chain.handle({"token": "t", "body": {}})        # -> None (passed every check)
chain.handle({"body": {}})                      # -> '401 unauthorized'
chain.handle({"token": "t", "calls": 500})      # -> '429 too many requests'
```

### ATM dispenser (each handler partly handles)

```python
class NoteDispenser:
    def __init__(self, note: int, nxt: "NoteDispenser | None" = None) -> None:
        self.note, self.next = note, nxt

    def dispense(self, amount: int, out: dict[int, int]) -> dict[int, int]:
        count, rest = divmod(amount, self.note)
        if count:
            out[self.note] = count
        if rest and self.next:
            return self.next.dispense(rest, out)
        if rest:
            raise ValueError(f"cannot dispense {rest}")
        return out

atm = NoteDispenser(2000, NoteDispenser(500, NoteDispenser(100)))
atm.dispense(2800, {})          # -> {2000: 1, 500: 1, 100: 3}
```

### Logging levels

```python
from enum import IntEnum

class Level(IntEnum):
    DEBUG = 10
    INFO = 20
    ERROR = 40

class Logger:
    def __init__(self, level: Level, nxt: "Logger | None" = None) -> None:
        self.level, self.next = level, nxt

    def log(self, level: Level, msg: str) -> None:
        if level >= self.level:                 # IntEnum compares like int
            self.write(msg)
        if self.next:
            self.next.log(level, msg)           # every eligible logger writes

    def write(self, msg: str) -> None: ...

class ConsoleLogger(Logger):
    def write(self, msg: str) -> None: print(f"console: {msg}")

class ErrorFileLogger(Logger):
    def write(self, msg: str) -> None: print(f"error file: {msg}")

log = ConsoleLogger(Level.DEBUG, ErrorFileLogger(Level.ERROR))
log.log(Level.INFO, "started")      # console only
log.log(Level.ERROR, "crash")       # console + error file
```

### List of callables (Pythonic pipeline)

```python
from typing import Callable

type Check = Callable[[dict], str | None]       # 3.12+; returns error or None

def has_user(o: dict) -> str | None:
    return None if o.get("user") else "missing user"

def has_items(o: dict) -> str | None:
    return None if o.get("items") else "empty cart"

def in_stock(o: dict) -> str | None:
    return None if all(i != "sold-out" for i in o["items"]) else "out of stock"

CHECKS: list[Check] = [has_user, has_items, in_stock]   # order matters

def validate(order: dict) -> str | None:
    for check in CHECKS:
        if (err := check(order)) is not None:   # first failure stops the chain
            return err
    return None

validate({"user": "amy", "items": ["tea"]})     # -> None
validate({"user": "amy", "items": []})          # -> 'empty cart'
```

### Approval escalation

```python
APPROVERS = [("team_lead", 1_000), ("manager", 10_000), ("director", 100_000)]

def approver_for(amount: int) -> str:
    return next((who for who, limit in APPROVERS if amount <= limit), "ceo")

approver_for(5_000)             # -> 'manager'
approver_for(500_000)           # -> 'ceo'
```
